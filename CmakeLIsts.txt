cmake_minimum_required(VERSION 3.10)

project(DormitoryManagement LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

set(FETCHCONTENT_UPDATES_DISCONNECTED ON)
set(FETCHCONTENT_QUIET OFF)

FetchContent_Declare(ftxui
  GIT_REPOSITORY https://github.com/arthursonzogni/ftxui.git
  GIT_TAG        v5.0.0
  GIT_PROGRESS   TRUE
  GIT_SHALLOW    TRUE
  EXCLUDE_FROM_ALL
)
FetchContent_MakeAvailable(ftxui)

add_subdirectory(src)

add_executable(dormitory
    main.cpp
    src/ui/ui.cpp
    src/ui/table.cpp
    src/ui/Dashboard.cpp
    src/ui/events/common_events.cpp
    src/ui/events/search_event.cpp
    src/ui/events/action_event.cpp
    src/ui/events/student_event.cpp
    src/ui/events/room_event.cpp
    src/ui/events/contract_event.cpp
    src/ui/events/invoice_event.cpp
    src/ui/events/report_event.cpp
)

target_link_libraries(dormitory PRIVATE
    dormitory_lib
    ftxui::screen
    ftxui::dom
    ftxui::component
)

target_include_directories(dormitory PRIVATE
    ${CMAKE_SOURCE_DIR}/src
)

set_target_properties(dormitory PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/output
)

target_compile_definitions(dormitory PRIVATE
    "DATA_DIR_PATH=\"${CMAKE_SOURCE_DIR}/data\""
)
